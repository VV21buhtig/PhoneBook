## FILE 3: Contact.cs

<a id='contactcs'></a>

```csharp
﻿using PhoneBook.ViewModels;
using System.ComponentModel;
using System.Runtime.CompilerServices;
using System.Text.RegularExpressions;

namespace PhoneBook.Models
{
    public class Contact : ObservableObject
    {
        private string _name = string.Empty;
        private string _phone = string.Empty;

        public Contact(string name, string phone)
        {
            Name = name;
            Phone = phone;
            Validate();
        }

        public string Name
        {
            get => _name;
            set
            {
                if (Set(ref _name, value))
                    Validate();
            }
        }

        public string Phone
        {
            get => _phone;
            set
            {
                if (Set(ref _phone, value))
                    Validate();
            }
        }

        public bool IsValid { get; private set; } = true;
        public string? ValidationError { get; private set; }

        private void Validate()
        {
            if (string.IsNullOrWhiteSpace(Name))
            {
                IsValid = false;
                ValidationError = "Имя не может быть пустым";
            }
            else if (!Regex.IsMatch(Phone, @"^(\+7|8)?\d{10}$"))
            {
                IsValid = false;
                ValidationError = "Телефон должен быть в формате +7XXXXXXXXXX или 8XXXXXXXXXX";
            }
            else
            {
                IsValid = true;
                ValidationError = null;
            }
        }
    }
}
```

## FILE 5: MainViewModel.cs

<a id='mainviewmodelcs'></a>

```csharp
﻿using System.Collections.ObjectModel;
using System.Windows.Input;
using PhoneBook.Models;

namespace PhoneBook.ViewModels
{
    public class MainViewModel : ObservableObject
    {
        public ObservableCollection<Contact> Contacts { get; }

        private string _name = string.Empty;
        public string Name
        {
            get => _name;
            set => Set(ref _name, value);
        }

        private string _phone = string.Empty;
        public string Phone
        {
            get => _phone;
            set => Set(ref _phone, value);
        }

        private Contact? _selectedContact;
        public Contact? SelectedContact
        {
            get => _selectedContact;
            set => Set(ref _selectedContact, value);
        }

        private string _inputError = string.Empty;
        public string InputError
        {
            get => _inputError;
            set => Set(ref _inputError, value);
        }

        public ICommand AddCommand { get; }
        public ICommand DeleteCommand { get; }

        public MainViewModel()
        {
            Contacts = new ObservableCollection<Contact>();
            AddCommand = new RelayCommand(AddContact);
            DeleteCommand = new RelayCommand<Contact?>(DeleteContact);
        }

        private void AddContact()
        {
            var contact = new Contact(Name, Phone);

            if (!contact.IsValid)
            {
                InputError = contact.ValidationError;
                return;
            }

            InputError = string.Empty;
            Contacts.Add(contact);
            Name = string.Empty;
            Phone = string.Empty;
        }

        private void DeleteContact(Contact? contact)
        {
            if (contact != null && Contacts.Contains(contact))
            {
                Contacts.Remove(contact);
            }
        }
    }
}
```

## FILE 6: ObservableObject.cs

<a id='observableobjectcs'></a>

```csharp
﻿using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace PhoneBook.ViewModels
{
    public abstract class ObservableObject : INotifyPropertyChanged
    {
        public event PropertyChangedEventHandler? PropertyChanged;

        protected virtual void OnPropertyChanged([CallerMemberName] string? propertyName = null)
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
        }

        protected bool Set<T>(ref T field, T value, [CallerMemberName] string? propertyName = null)
        {
            if (EqualityComparer<T>.Default.Equals(field, value))
                return false;

            field = value;
            OnPropertyChanged(propertyName);
            return true;
        }
    }
}
```

## FILE 7: RelayCommand.cs

<a id='relaycommandcs'></a>

```csharp
﻿using System.Windows.Input;

namespace PhoneBook.ViewModels
{
    public class RelayCommand(Action execute, Func<bool>? canExecute = null) : ICommand
    {
        private readonly Action _execute = execute;
        private readonly Func<bool>? _canExecute = canExecute;

        public bool CanExecute(object? parameter) => _canExecute?.Invoke() ?? true;

        public void Execute(object? parameter)
        {
            if (CanExecute(parameter))
                _execute.Invoke();
        }

        public event EventHandler? CanExecuteChanged
        {
            add => CommandManager.RequerySuggested += value;
            remove => CommandManager.RequerySuggested -= value;
        }
    }

    public class RelayCommand<T>(Action<T?> execute, Predicate<T?>? canExecute = null) : ICommand
    {
        private readonly Action<T?> _execute = execute ?? throw new ArgumentNullException(nameof(execute));
        private readonly Predicate<T?>? _canExecute = canExecute;

        public bool CanExecute(object? parameter) => _canExecute?.Invoke((T?)parameter) ?? true;

        public void Execute(object? parameter)
        {
            if (CanExecute(parameter))
                _execute.Invoke((T?)parameter);
        }

        public event EventHandler? CanExecuteChanged
        {
            add => CommandManager.RequerySuggested += value;
            remove => CommandManager.RequerySuggested -= value;
        }
    }
}
```

## Контрольные вопросы по лабораторной работе №9 (MVVM)

### Вопрос 1: Что такое MVVM и для чего он нужен?

MVVM (Model-View-ViewModel) — это архитектурный паттерн, который **разделяет приложение на три независимые части**: модель (данные и бизнес-логика), представление (интерфейс) и модель представления (посредник). Он был разработан Microsoft в 2005 году специально для WPF.

**Зачем нужен:**
- Чтобы **отделить интерфейс от логики** — интерфейс может меняться без изменения кода
- Чтобы **упростить тестирование** — ViewModel можно тестировать без запуска окна
- Чтобы **разработчики и дизайнеры работали параллельно** — одни пишут логику, другие верстают XAML
- Чтобы **автоматически синхронизировать данные** между интерфейсом и кодом через механизм привязок (binding)

Без MVVM вся логика валилась бы в code-behind (MainWindow.xaml.cs), что делало код нечитаемым, нетестируемым и трудно поддерживаемым.

---

### Вопрос 2: Назовите и охарактеризуйте каждого участника MVVM

**Model (Модель)**
- Хранит бизнес-данные и бизнес-логику
- Реализует `INotifyPropertyChanged`, чтобы уведомлять об изменениях свойств
- Ничего не знает о View и ViewModel (полностью независима)
- *В вашем коде:* `Contact.cs`

**View (Представление)**
- Это XAML-разметка (окно, страница, пользовательский элемент)
- Отвечает только за отображение данных и визуальное взаимодействие
- Не содержит бизнес-логики (только привязки и команды)
- Устанавливает `DataContext` (ссылку на ViewModel)
- *В вашем коде:* `MainWindow.xaml`

**ViewModel (Модель представления)**
- Связующее звено между Model и View
- Предоставляет свойства для привязки (через `INotifyPropertyChanged`)
- Предоставляет команды (через `ICommand`) для действий пользователя
- Хранит коллекции (`ObservableCollection`) для автоматического обновления UI
- Ничего не знает о View (нет ссылок на кнопки, текстбоксы и т.д.)
- *В вашем коде:* `MainViewModel.cs`

**Дополнительные инфраструктурные классы**
- `ObservableObject` — базовая реализация `INotifyPropertyChanged` с методом `Set`
- `RelayCommand` — реализация `ICommand` для обёртки методов ViewModel в команды

---

### Вопрос 3: Какие интерфейсы и классы лежат в основе MVVM в WPF и за что каждый отвечает?

**INotifyPropertyChanged**
- Интерфейс из пространства `System.ComponentModel`
- Позволяет объекту уведомлять UI об изменении значений свойств
- Без него привязка (Binding) не узнает, что данные обновились
- Имеет одно событие: `PropertyChanged`

**ICommand**
- Интерфейс из `System.Windows.Input`
- Позволяет связать действие пользователя (нажатие кнопки) с методом в ViewModel
- Содержит методы `Execute()` (что делать) и `CanExecute()` (можно ли выполнить)
- Имеет событие `CanExecuteChanged` для включения/отключения кнопок

**ObservableCollection<T>**
- Специальная коллекция из `System.Collections.ObjectModel`
- Автоматически уведомляет UI об изменениях (добавление, удаление, перемещение элементов)
- Обычная коллекция (List, array) этого не умеет — интерфейс не обновится

**Классы из вашей реализации:**
- `ObservableObject` — базовый класс, реализующий `INotifyPropertyChanged` с удобным методом `Set<T>`
- `RelayCommand` — реализует `ICommand`, оборачивает `Action` (для команд без параметра) и `Action<T>` (для команд с параметром)
- `CommandManager.RequerySuggested` — событие, которое автоматически перепроверяет состояние `CanExecute` (например, при изменении фокуса, вводе текста)

---

### Вопрос 4: Как в вашем приложении реализована привязка данных (Data Binding) между View и ViewModel?

**Установка DataContext**

В `MainWindow.xaml` есть строка:
```xml
<Window.DataContext>
    <vm:MainViewModel/>
</Window.DataContext>
```
Это создаёт экземпляр ViewModel и назначает его контекстом данных для всего окна.

**Привязка свойства Text文本框**

```xml
<TextBox Text="{Binding Name, UpdateSourceTrigger=PropertyChanged}"/>
```
- `Name` — это имя свойства в `MainViewModel` (public string Name)
- `UpdateSourceTrigger=PropertyChanged` — обновляет ViewModel при каждом нажатии клавиши (не только при потере фокуса)
- Работает в обе стороны: пользователь вводит → меняется свойство в ViewModel; код меняет свойство → обновляется текст в текстовом поле

**Привязка списка**

```xml
<DataGrid ItemsSource="{Binding Contacts}" SelectedItem="{Binding SelectedContact}">
```
- `ItemsSource` связывает коллекцию `ObservableCollection<Contact>` из ViewModel с таблицей
- `SelectedItem` связывает выбранную строку со свойством `SelectedContact` — это двусторонняя привязка, позволяющая удалять выбранный элемент

**Привязка команд**

```xml
<Button Command="{Binding AddCommand}"/>
<Button Command="{Binding DeleteCommand}" CommandParameter="{Binding SelectedContact}"/>
```
- `Command` связывает кнопку с командой из ViewModel (`AddCommand` или `DeleteCommand`)
- `CommandParameter` передаёт параметр в команду (в данном случае — выбранный контакт)

---

### Вопрос 5: Что даёт использование `ObservableCollection<T>` вместо обычной коллекции?

**ObservableCollection<T> автоматически уведомляет UI об изменениях**:

`ObservableCollection<Contact>` реализует интерфейсы `INotifyCollectionChanged` и `INotifyPropertyChanged`. Когда вы вызываете `Contacts.Add(contact)` или `Contacts.Remove(contact)`, коллекция сама генерирует событие. WPF подписан на это событие и обновляет DataGrid/ListView без вашего участия.

Если бы вы использовали `List<Contact>`:
- После добавления элемента таблица осталась бы пустой
- Пришлось бы вручную переприсваивать свойство (`Contacts = new List...`) и вызывать `PropertyChanged`
- Код стал бы грязным и нетестируемым

**В вашем коде:**
```csharp
public ObservableCollection<Contact> Contacts { get; }
```
Инициализируется в конструкторе один раз. Все операции добавления и удаления напрямую обновляют UI.

### Вопрос 6: Как работает команда `RelayCommand`? Почему кнопка «Удалить» становится неактивной без выбранного контакта?

**RelayCommand — это обёртка, превращающая обычный метод в команду.**

Интерфейс `ICommand` требует два метода:
- `Execute(object parameter)` — что делать
- `CanExecute(object parameter)` — можно ли выполнить (возвращает true/false)

В вашем коде:
```csharp
DeleteCommand = new RelayCommand<Contact?>(DeleteContact);
```

Конструктор `RelayCommand<T>`:
- Первый параметр — метод, который будет вызван в `Execute`
- Второй параметр (опциональный) — функция-предикат для `CanExecute`

**Почему вы не передали canExecute для DeleteCommand?**

Потому что `CanExecute` по умолчанию возвращает `true` если переданный параметр не null. Класс `RelayCommand<T>` имеет реализацию:
```csharp
public bool CanExecute(object? parameter) => _canExecute?.Invoke((T?)parameter) ?? true;
```
Если `_canExecute` не задан, возвращается `true`. Но здесь нет проверки на null? На самом деле проверка происходит за счёт того, что `_execute` выбрасывает исключение, если параметр null? Нет. Посмотрите внимательнее на `Execute` в `RelayCommand<T>`:

```csharp
public void Execute(object? parameter)
{
    if (CanExecute(parameter))
        _execute.Invoke((T?)parameter);
}
```
`_execute.Invoke((T?)parameter)` — если `parameter` null, а команда ожидает `Contact?`, то вызовется `DeleteContact(null)`. В теле `DeleteContact` есть проверка:
```csharp
if (contact != null && Contacts.Contains(contact))
```
Так что `null` просто ничего не делает. Но кнопка остаётся активной даже без выбранного контакта — это **минус**. Правильнее было бы добавить `canExecute`:

```csharp
DeleteCommand = new RelayCommand<Contact?>(DeleteContact, c => c != null);
```

Тогда кнопка автоматически отключалась бы, когда `SelectedContact == null`. Без этого кнопка активна всегда, просто ничего не делает.

**Обновление состояния кнопки:**
Событие `CanExecuteChanged` привязано к `CommandManager.RequerySuggested`. WPF автоматически вызывает `CanExecute` при изменении фокуса, вводе данных, нажатии клавиш. Поэтому при смене выбранного контакта кнопка перепроверяет своё состояние.

---

### Вопрос 7: Где выполняются методы `AddContact()` и `DeleteContact()`? Почему в `MainViewModel`, а не в `MainWindow.xaml.cs`?

Методы находятся в `MainViewModel.cs`, а не в code-behind, потому что:

**Это бизнес-логика, а не логика представления.**

- `AddContact()` создаёт новый объект `Contact`, проверяет валидность, добавляет в коллекцию, очищает поля ввода
- `DeleteContact()` удаляет выбранный контакт из коллекции

Если бы эти методы были в `MainWindow.xaml.cs`:
1. Их нельзя было бы протестировать без запуска окна
2. Пришлось бы лезть в код интерфейса, чтобы менять бизнес-правила
3. Замена интерфейса (например, с WPF на другое) потребовала бы переписывания всей логики
4. Нарушился бы принцип единственной ответственности — окно отвечало бы и за отображение, и за обработку данных

В `MainViewModel` эти методы легко тестируются через юнит-тесты и не зависят от того, какой именно интерфейс их вызывает.

### Вопрос 8: Как в вашем приложении реализована валидация данных и где она находится?

**Валидация находится в модели `Contact.cs`, а не в ViewModel.**

Метод `Validate()` вызывается в конструкторе и в сеттерах свойств `Name` и `Phone`:

```csharp
public string Name
{
    get => _name;
    set
    {
        if (Set(ref _name, value))
            Validate();  // ← валидация при каждом изменении
    }
}
```

**Правила валидации:**
1. Имя не должно быть пустым или состоять из пробелов (`string.IsNullOrWhiteSpace`)
2. Телефон должен соответствовать регулярному выражению `^(\+7|8)?\d{10}$`:
   - Может начинаться с +7 (например, +79161234567)
   - Может начинаться с 8 (например, 89161234567)
   - Может быть без кода страны (10 цифр)
   - Всего 10 цифр после кода

**Результаты валидации:**
- `IsValid` — true/false для проверки перед добавлением
- `ValidationError` — текст ошибки, который показывается пользователю

**Где используется в ViewModel:**
```csharp
private void AddContact()
{
    var contact = new Contact(Name, Phone);
    if (!contact.IsValid)
    {
        InputError = contact.ValidationError;  // ← показываем ошибку
        return;
    }
    InputError = string.Empty;
    Contacts.Add(contact);
    // ...
}
```

**В XAML** ошибка отображается в красном `TextBlock`:
```xml
<TextBlock Grid.Row="2" Foreground="Red" Text="{Binding InputError}" />
```








## Контрольные вопросы по MVVM (КРАТКО)

---

**1. Что такое MVVM и зачем нужен?**

MVVM (Model-View-ViewModel) — архитектурный паттерн для разделения интерфейса, данных и логики. Нужен, чтобы код был тестируемым, поддерживаемым и позволял работать дизайнерам и программистам параллельно.

---

**2. Участники MVVM**

- **Model** — данные и бизнес-логика. Не знает о View и ViewModel.
- **View** — XAML-интерфейс. Только отображение, без логики.
- **ViewModel** — посредник. Свойства (`INotifyPropertyChanged`), команды (`ICommand`), коллекции (`ObservableCollection`). Не знает о View.

---

**3. Базовые интерфейсы и классы**

- `INotifyPropertyChanged` — уведомляет UI об изменении свойств
- `ICommand` — обёртка действий пользователя (кнопки)
- `ObservableCollection<T>` — коллекция с автообновлением UI
- `ObservableObject` — базовая реализация `INotifyPropertyChanged`
- `RelayCommand` — реализация `ICommand`

---

**4. Data Binding**

- `DataContext` устанавливает ViewModel как источник данных
- `Text="{Binding Name}"` — привязка свойства
- `ItemsSource="{Binding Contacts}"` — привязка коллекции
- `Command="{Binding AddCommand}"` — привязка команды

---

**5. ObservableCollection**

Автоматически уведомляет UI при добавлении/удалении элементов. Обычный `List` этого не умеет — интерфейс не обновится.

---

**6. RelayCommand**

Оборачивает метод в команду. `Execute` — что делать, `CanExecute` — доступна ли кнопка. Событие `CanExecuteChanged` перепроверяет состояние кнопки.

---

**7. Где логика (AddContact, DeleteContact)?**

В `MainViewModel`, а не в code-behind. Чтобы можно было тестировать без запуска окна, менять интерфейс независимо и не нарушать разделение ответственности.

---

**8. Валидация**

В модели `Contact`. При изменении `Name` или `Phone` вызывается `Validate()`. Проверяет:
- Имя не пустое
- Телефон в формате `+7XXXXXXXXXX`, `8XXXXXXXXXX` или 10 цифр

Результат — `IsValid` и `ValidationError`.









## Контрольные вопросы по MVVM (без кода, чистая теория)

---

**1. Что такое MVVM и для чего он нужен?**

MVVM (Model-View-ViewModel) — это архитектурный паттерн, который разделяет приложение на три части: данные (модель), интерфейс (представление) и посредника между ними (модель представления).
Он нужен для того, чтобы отделить бизнес-логику от интерфейса, упростить тестирование, позволить дизайнерам и разработчикам работать параллельно,
а также автоматически синхронизировать данные между кодом и интерфейсом через механизм привязок.

---

**2. Назовите и охарактеризуйте каждого участника MVVM**

- **Model (Модель)** — содержит данные и бизнес-логику. Ничего не знает о View и ViewModel, полностью независима. Уведомляет об изменениях своих свойств.

- **View (Представление)** — это интерфейс пользователя (XAML). Отвечает только за отображение и не содержит бизнес-логики. Связывается с ViewModel через механизм привязки данных.

- **ViewModel (Модель представления)** — посредник между Model и View. Предоставляет свойства для отображения, команды для действий пользователя и коллекции для списков. 
- ьНе знает о View — нет прямых ссылок на кнопки, поля и т.д.

---

**3. Какие интерфейсы лежат в основе MVVM в WPF и за что каждый отвечает?**

- **INotifyPropertyChanged** — позволяет объекту уведомлять интерфейс об изменении значений свойств. Без него привязка не узнает, что данные обновились.

- **ICommand** — позволяет связать действие пользователя (нажатие кнопки) с методом в ViewModel. Предоставляет методы Execute (что делать) и CanExecute (можно ли выполнить).

- **ObservableCollection** — специальная коллекция, которая автоматически уведомляет интерфейс о добавлении или удалении элементов. Обычный список этого не умеет.

---

**4. Как реализуется привязка данных между View и ViewModel?**

View устанавливает свой DataContext (контекст данных) на экземпляр ViewModel. После этого элементы интерфейса через Binding (привязку) указывают, 
какое свойство или команду из ViewModel они используют. Привязка может быть двусторонней: изменения в интерфейсе автоматически попадают в ViewModel, а изменения в ViewModel — в интерфейс.

---

**5. Почему используют ObservableCollection вместо обычных коллекций?**

ObservableCollection автоматически уведомляет UI об изменениях — когда вы добавляете или удаляете элемент, интерфейс обновляется сам.
Обычные коллекции (List, Array) этого не умеют, поэтому интерфейс не увидит изменения — пришлось бы пересоздавать всю коллекцию и уведомлять вручную.

---

**6. Как работает RelayCommand и как кнопка становится неактивной?**

RelayCommand — это обёртка, которая превращает обычный метод в команду. Он хранит два делегата: Execute (что делать) и CanExecute (можно ли выполнить). 
Кнопка вызывает CanExecute перед отображением и блокируется, если тот вернул false. WPF автоматически перепроверяет CanExecute через CommandManager, когда происходит какое-либо действие в интерфейсе.

---

**7. Где должна находиться бизнес-логика (методы добавления/удаления) — в ViewModel или в code-behind? Почему?**

В ViewModel. Потому что это логика работы с данными, а не отображения. Если положить её в code-behind (файл окна), то:
- её нельзя покрыть юнит-тестами без запуска окна
- при смене интерфейса придётся переписывать логику
- нарушается принцип разделения ответственности

ViewModel можно тестировать изолированно, она не зависит от того, какое именно окно её использует.

---

**8. Где должна находиться валидация данных и почему?**

В модели (Model). Валидация касается правильности самих данных (имя не пустое, телефон в правильном формате) — это бизнес-правила, а не логика отображения.
ViewModel только проверяет результат валидации (IsValid) и показывает ошибку, но сами правила хранятся в модели. Это позволяет переиспользовать модель в других частях приложения с теми же правилами.
